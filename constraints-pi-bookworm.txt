# Raspberry Pi OS Bookworm (Debian 12) only; current Pi OS is Trixie and needs none of this.
#
#   pip install -c constraints-pi-bookworm.txt .
#
# Bookworm's python3-picamera2 from apt is built against the system numpy 1.24.
# OpenCV 4.12 and later need numpy 2, and installing that into the venv can
# break picamera2's compiled parts. 4.11.0.86 is the last OpenCV that works with
# numpy 1.x, and it has ARM64 wheels. These pins only apply as constraints: they
# fix versions without adding anything to install.
opencv-contrib-python==4.11.0.86
numpy>=1.23.5,<2
